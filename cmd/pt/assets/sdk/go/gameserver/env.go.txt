package gameserver

import (
	"crypto/ed25519"
	"encoding/base64"
	"errors"
	"fmt"
	"os"
	"strings"

	playtogether "github.com/Yanghsuanming/play-together/sdk-go"
)

// PlatformFromEnv 依環境變數建立玩聚平台的 client：
//
//	PT_API_URL   平台網址（pt dev 會設成本機假平台）
//	PT_KEY_ID    伺服器金鑰 ID（玩聚登記的）
//	PT_KEY_FILE  pt keygen 產生的私鑰檔（一行 base64 種子），預設 ../keys/server.key
//	PT_PRIVATE_KEY 也可以直接放 base64 種子（優先於 PT_KEY_FILE）
//	PT_MOCK=1    舊的模擬模式：不連平台（只限本機開發）
func PlatformFromEnv() (*playtogether.Client, error) {
	c := &playtogether.Client{BaseURL: os.Getenv("PT_API_URL"), KeyID: os.Getenv("PT_KEY_ID"), Mock: os.Getenv("PT_MOCK") == "1"}
	if c.Mock {
		return c, nil
	}
	if c.BaseURL == "" || c.KeyID == "" {
		return nil, errors.New("請設定 PT_API_URL 與 PT_KEY_ID（本機開發用 pt dev 會自動設定）")
	}
	raw := strings.TrimSpace(os.Getenv("PT_PRIVATE_KEY"))
	if raw == "" {
		file := os.Getenv("PT_KEY_FILE")
		if file == "" {
			file = "../keys/server.key"
		}
		b, err := os.ReadFile(file)
		if err != nil {
			return nil, fmt.Errorf("讀不到伺服器私鑰（PT_KEY_FILE）：%w", err)
		}
		raw = strings.TrimSpace(string(b))
	}
	seed, err := base64.StdEncoding.DecodeString(raw)
	if err != nil || len(seed) != ed25519.SeedSize {
		return nil, errors.New("伺服器私鑰格式錯誤：請用 pt keygen -purpose server 產生")
	}
	c.PrivateKey = ed25519.NewKeyFromSeed(seed)
	return c, nil
}
